**Step 3: Create `src/features/wallet/hooks/use-wallet-connection.ts`.**

Wraps `useAppKit` (for `open`/`disconnect`) and reads from Zustand store for status/error. IMPORTANT: use `disconnect()` from `useAppKit()` — `useDisconnect()` is deprecated. Manage connecting state manually in Zustand, NOT via `isLoading` from `useAppKitState` (known bug #4677).

```typescript
import { useAppKit } from '@reown/appkit-react-native';
import { useWalletStore } from '../stores/wallet-store';

export function useWalletConnection() {
  const { open, disconnect } = useAppKit();
  const { address, isConnected, status, error, setStatus, setError } =
    useWalletStore();

  const connect = async () => {
    try {
      setStatus('connecting');
      setError(null);
      await open();
      // Status transitions to 'connected' via useWalletSync when address arrives
    } catch (e) {
      setStatus('error');
      setError(e instanceof Error ? e.message : 'Connection failed');
    }
  };

  const handleDisconnect = async () => {
    try {
      await disconnect();
      // useWalletSync handles clearing the store
    } catch {
      // Swallow disconnect errors (known issue #13 in some versions)
      // useWalletSync will still clear the store if session ends
    }
  };

  const cancelConnection = () => {
    setStatus('idle');
    setError(null);
  };

  return {
    address,
    isConnected,
    status,
    error,
    connect,
    disconnect: handleDisconnect,
    cancelConnection,
  };
}
```

**Step 4: Create `src/features/wallet/hooks/use-wallet-sync.ts`.**

Called exactly once inside AppProvider via WalletSyncBridge. Syncs `useAppKitAccount` state to Zustand. Also watches `useAppKitState().isOpen` to detect modal dismissal without connection (resets stuck 'connecting' status).

```typescript
import { useEffect } from 'react';
import { useAppKitAccount, useAppKitState } from '@reown/appkit-react-native';
import { useWalletStore } from '../stores/wallet-store';

export function useWalletSync() {
  const { address, isConnected } = useAppKitAccount();
  const { isOpen } = useAppKitState();
  const { setWallet, clearWallet, status, setStatus } = useWalletStore();

  // Sync AppKit account state to Zustand
  useEffect(() => {
    if (isConnected && address) {
      setWallet({ address, isConnected: true });
    } else if (!isConnected) {
      clearWallet();
    }
  }, [address, isConnected]);

  // Detect modal dismissed without connecting (isLoading bug workaround)
  useEffect(() => {
    if (!isOpen && status === 'connecting' && !isConnected) {
      setStatus('idle');
    }
  }, [isOpen]);
}
```
