```typescript
export interface IEthereumProvider {
    getGasPrice(): Promise&lt;string&gt;;
    getBlockNumber(): Promise&lt;string&gt;;
    getBalance(address: string): Promise&lt;string&gt;;
}

export interface ICacheStore {
    get(key: string): Promise&lt;string | null&gt;;
    set(
        key: string,
        value: string,
        ttlSeconds: number,
    ): Promise&lt;void&gt;;
}

export interface BalanceSaveDto {
    address: string;
    balanceWei: string;
    fetchedAt: Date;
}

export interface IBalanceRepository {
    save(data: BalanceSaveDto): Promise&lt;void&gt;;
}
```

Three port interfaces per user decision. `IEthereumProvider` has 3 methods mapping 1:1 to Etherscan endpoints. `ICacheStore` is generic (reusable). `IBalanceRepository` uses plain DTO (not TypeORM entity). `BalanceSaveDto` is co-located with the repository interface it serves.

File 1: `src/component/ethereum/errors.ts` (ETH-03)

```typescript
export class ValidationError extends Error {
    public readonly code = 'VALIDATION_ERROR' as const;

    constructor(message: string) {
        super(message);
        this.name = 'ValidationError';
        Object.setPrototypeOf(this, ValidationError.prototype);
    }
}

export class EtherscanApiError extends Error {
    public readonly code = 'UPSTREAM_ERROR' as const;

    constructor(message: string) {
        super(message);
        this.name = 'EtherscanApiError';
        Object.setPrototypeOf(this, EtherscanApiError.prototype);
    }
}
```

Both classes extend Error with `Object.setPrototypeOf` per research (Pitfall 2 -- required for correct `instanceof` in TypeScript CJS output). The `code` property is a `const` assertion matching the `ErrorBody.code` union from response-models.ts. `ValidationError` maps to HTTP 400 (CORE-05). `EtherscanApiError` maps to HTTP 502 (ETH-03). The controller (Phase 4) will use `instanceof` to distinguish them.
