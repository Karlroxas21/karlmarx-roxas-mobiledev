1. **src/wire.ts** — Add imports for TypeORM DataSource and ioredis Redis, then initialize both in the `// Infra` block. The complete file should be:

   ```typescript
   import express from 'express';
   import { DataSource } from 'typeorm';
   import { Redis } from 'ioredis';
   import { Server, Controller } from './server';
   import { config, logger } from './config';

   export const createServer = async (): Promise<Server> => {
       const app = express();

       // Infra
       const dataSource = new DataSource({
           type: 'postgres',
           url: config.databaseUrl,
           synchronize: process.env.NODE_ENV !== 'production',
           logging: process.env.NODE_ENV === 'development',
           entities: [],
       });
       await dataSource.initialize();
       logger.info('PostgreSQL DataSource initialized');

       const redis = new Redis(config.redisUrl);
       redis.on('error', (err: Error) => {
           logger.error('Redis client error', { error: err.message });
       });
       logger.info('ioredis client created');

       // Services

       // Controllers
       const controllers: Controller[] = [];

       const server = new Server(
           app,
           controllers,
           config.hostname,
           config.port,
       );

       logger.info(`server wired on ${config.hostname}:${config.port}`);

       return server;
   };
   ```

   Key details:
   - `synchronize: process.env.NODE_ENV !== 'production'` — auto-sync schema in dev, NEVER in production
   - `logging: process.env.NODE_ENV === 'development'` — SQL logging only in dev
   - `entities: []` — empty for Phase 1; entities added in Phase 3
   - `await dataSource.initialize()` — must be awaited; constructor alone does NOT connect
   - `redis.on('error', ...)` — MUST be attached immediately after construction; prevents unhandled error crashes when Redis is unreachable
   - Do NOT call `process.exit` in the Redis error handler (ioredis auto-reconnects; transient blips should not kill the process)
   - Use named import `{ Redis }` (not `import Redis from 'ioredis'` or `import * as Redis`) — ioredis v5 pattern
